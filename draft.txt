const getCoordinates = async (street, number, city) => {
    
    const fullAddress = `${number}%20${street},%20${city},%20Greece`;
    // const encodedAddress = encodeURIComponent(fullAddress);
    const apiUrl = `https://nominatim.openstreetmap.org/search?q=${fullAddress}&format=json&limit=1`;
    console.log(`Nominatim API URL: ${apiUrl}`); // Log the full URL
    try {
        // Use node-fetch to make the API call
        const fetch = (await import('node-fetch')).default;
        const response = await fetch(apiUrl, {
            headers: {
                'User-Agent': 'Web_2024/1.0' // Customize the User-Agent string
            }
        });
        const data = await response.json();

        if (data && data.length > 0) {
            const { lon, lat } = data[0]; // Nominatim returns lon/lat
            return { lat: parseFloat(lat), lng: parseFloat(lon) };
        } else {
            throw new Error('Could not retrieve coordinates from Nominatim.');
        }
    } catch (error) {
        console.error('Error in Nominatim geocoding:', error.message);
        throw error; // Re-throw the error to handle it in the calling function
    }
};

    try {
        console.log(`Finding user with username or email: ${login}`);
        // Look for the user in the base model (User) which includes all roles
        const user = await User.findOne({
            $or: [{ username: login }, { email: login }] // Search by either username or email
        });
        
        console.log(user);
        
        // If user is not found, return error
        if (!user) {
            return res.render('login', { error: 'Invalid username or password.' });
        }

        // Check if the entered password matches the hashed password
        const isPasswordCorrect = await user.correctPassword(password, user.password);
        if (!isPasswordCorrect) {
            return res.render('login', { error: 'Invalid username or password.' });
        }
        req.session.userId = user._id;
        // Redirect based on the user's role
        if (user.role === 'civilian') {
            // Redirect to civilian dashboard
            return res.redirect('/civilianDashboard');
        } else if (user.role === 'rescuer') {
            // Redirect to rescuer dashboard
            return res.redirect('/rescuerDashboard');
        } else if (user.role === 'admin') {
            // Redirect to admin dashboard
            return res.redirect('/adminDashboard');
        }

        // Handle case if no matching role is found
        return res.render('login', { error: 'User role not recognized.' });

    } catch (err) {
        // Handle any errors
        return res.render('login', { error: 'Error in login: ' + err.message });
    }
});

const reverseGeocodeNominatim = async (lat, lng) => {
    const apiUrl = `https://nominatim.openstreetmap.org/reverse?lat=${lat}&lon=${lng}&format=json&addressdetails=1`;
    console.log(`apiURl: ${apiUrl}`);
    
    try {
        // Use node-fetch to make the API call
        const fetch = (await import('node-fetch')).default;
        const response = await fetch(apiUrl, {
            headers: {
                'User-Agent': 'YourAppNameHere/1.0' // Customize the User-Agent string
            }
        });
        const data = await response.json();

        if (data && data.address) {
            // Extract address details from the response
            const { road, house_number, city, postcode, country } = data.address;

            // Return available fields or default to null
            return {
                street: road || null,
                number: house_number || null,
                city: city || null,
                zip: postcode || null,
                country: country || null
            };
        } else {
            throw new Error('No valid address found');
        }
    } catch (error) {
        console.error('Error in Nominatim reverse geocoding:', error.message);
        return {
            street: null,
            number: null,
            city: null,
            zip: null,
            country: null
        }; // Return null for all fields if geocoding fails

};

    }

    // Map items to include category names and formatted details
    const itemsWithCategories = data.items.map(item => {
      // Find the category by its ID
      const category = data.categories.find(cat => cat.id === item.category);
      // Check for null/undefined and provide fallback values
      const formattedDetails = item.details && Array.isArray(item.details)
        ? item.details.map(detail => `${detail.detail_name || 'Unknown'}: ${detail.detail_value || 'N/A'}`).join(', ')
        : 'No details available';  // Default message for empty or null details

      return {
        id: item.id,
        name: item.name || 'Unnamed Item',  // Default name if null
        category: category ? category.category_name : 'No Category',  // Default if category not found
        details: formattedDetails,
        stock: item.stock !== undefined ? item.stock : 0  // Ensure stock is always a valid number
      };
    });

    res.json(itemsWithCategories);
  } catch (err) {
    console.error('Error during retrieval:', err);
    res.status(500).json({ error: err.message });
  }
});

    }

    const filePath = path.join(__dirname, '../uploads', req.file.filename);
    const fileData = fs.readFileSync(filePath, 'utf-8');

    // Parse JSON data
    const jsonData = JSON.parse(fileData);

    // Validate if JSON contains categories and items
    const { categories, items } = jsonData;
    if (!categories || !items) {
      return res.status(400).json({ error: 'Invalid JSON format. Must contain categories and items.' });
    }

    // Find the existing data document (assuming there's only one)
    let data = await Data.findOne();
    if (!data) {
      data = new Data({ categories: [], items: [] });
    }

    // Update or add categories
    categories.forEach(newCategory => {
      const existingCategory = data.categories.find(cat => cat.id === newCategory.id);
      if (existingCategory) {
        // Update existing category
        existingCategory.category_name = newCategory.category_name;
      } else {
        // Add new category
        data.categories.push(newCategory);
      }
    });

    // Update or add items
    items.forEach(newItem => {
      const existingItem = data.items.find(item => item.id === newItem.id);
      if (existingItem) {
        // Update existing item
        existingItem.name = newItem.name;
        existingItem.category = newItem.category;
        existingItem.details = newItem.details;
        existingItem.stock = newItem.stock;
      } else {
        // Add new item
        data.items.push(newItem);
      }
    });

    // Save the updated data to MongoDB
    await data.save();

    // Cleanup the uploaded file
    fs.unlinkSync(filePath);

    res.json({ message: 'Data imported successfully' });
  } catch (err) {
    console.error('Error during import:', err);
    res.status(500).json({ error: err.message });

// Utility function to calculate haversine distance between two coordinates
const haversineDistance = (lat1, lon1, lat2, lon2) => {
    const R = 6371e3; // Earth's radius in meters
    const φ1 = lat1 * Math.PI / 180;
    const φ2 = lat2 * Math.PI / 180;
    const Δφ = (lat2 - lat1) * Math.PI / 180;
    const Δλ = (lon2 - lon1) * Math.PI / 180;

    const a = Math.sin(Δφ / 2) * Math.sin(Δφ / 2) +
              Math.cos(φ1) * Math.cos(φ2) *
              Math.sin(Δλ / 2) * Math.sin(Δλ / 2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

    return R * c; // Distance in meters
};

        if (!dataDoc) {

        for (const obj of cargo) {
            console.log(`Processing cargo item: ${JSON.stringify(obj)}`);

            // Find the item by its id within the items array
            const item = dataDoc.items.find(i => i.id === obj.itemId);
            if (!item) {

            if (item.stock < obj.quantity) {
                return res.status(400).json({ success: false, message: `Insufficient stock for product: ${item.name}` });

            item.stock -= obj.quantity;

router.post('/unloadCargo', async (req, res) => {
    try {
        const userId = req.session.userId;
        const { items } = req.body;  // Array of items with itemName and quantity

        // Find the rescuer
        const user = await Rescuer.findById(userId);
        if (!user) {
            return res.status(404).json({ success: false, message: 'Rescuer not found' });
        }

        // Fetch the data document that contains all the items
        const dataDoc = await Data.findOne();
        if (!dataDoc) {
            return res.status(500).json({ success: false, message: 'Data document not found' });
        }

        // Iterate through the items to be unloaded
        for (const item of items) {
            const { itemName, quantity } = item;
            
            // Find the item in the general items collection by its name
            const itemInStorage = dataDoc.items.find(i => i.name === itemName);
            if (!itemInStorage) {
                return res.status(400).json({ success: false, message: `Item ${itemName} not found in storage` });
            }

            // Find the item in the rescuer's cargo
            const cargoItemIndex = user.cargo.findIndex(cargoItem => cargoItem.item === itemName);

            if (cargoItemIndex === -1) {
                return res.status(400).json({ success: false, message: `Item ${itemName} not found in rescuer's cargo` });
            }

            const cargoItem = user.cargo[cargoItemIndex];

            // Ensure enough quantity exists to unload
            if (cargoItem.quantity < quantity) {
                return res.status(400).json({ success: false, message: `Not enough quantity of ${itemName} in cargo to unload` });
            }

            // Update the stock in the general storage
            itemInStorage.stock += quantity;  // Add the unloaded quantity back to the stock

            // Deduct the quantity from the rescuer's cargo
            cargoItem.quantity -= quantity;

            // If the quantity becomes zero, remove the item from the rescuer's cargo
            if (cargoItem.quantity === 0) {
                user.cargo.splice(cargoItemIndex, 1);  // Remove the item from cargo
            }
        }

        // Save the updated rescuer data
        await user.save();

        // Save the updated stock back to the data document
        await dataDoc.save();

        res.json({ success: true, message: 'Cargo unloaded successfully, and stock updated.' });
    } catch (error) {
        console.error('Error unloading cargo:', error);
        res.status(500).json({ success: false, message: 'Internal Server Error' });
    }
});

        // Check how many tasks the rescuer currently has
        const rescuerTasksCount = rescuer.tasks.length;

        if (rescuerTasksCount >= 4) {
            return res.status(400).json({ message: 'You have already reached the limit of 4 tasks.' });
        }

        // Calculate how many more tasks can be added
        const remainingTaskSlots = 4 - rescuerTasksCount;

        if (taskIdsArray.length > remainingTaskSlots) {
            return res.status(400).json({ message: `You can only serve ${remainingTaskSlots} more task(s).` });
        }

        // Find the civilian by ID
        const civilian = await User.findById(civilianId);

        if (!civilian) {
            return res.status(404).json({ message: 'Civilian not found' });
        }

        // Loop through the civilian's tasks and update the matching ones
        civilian.tasks.forEach(task => {
            if (taskIdsArray.includes(task._id.toString())) { // Match tasks by _id
                task.rescuerUsername = rescuerId; // Update rescuer's ID
                task.dateServed = currentDate; // Update the served date
                task.status = 'processed'; // Update status to processed

                // Add the task to the rescuer's task list
                const taskCopy = {
                    ...task.toObject(), // Copy the task fields
                    rescuerUsername: rescuerId, // Make sure rescuerUsername is updated
                    dateServed: currentDate, // Ensure dateServed is current
                    status: 'processed' // Ensure status is 'processed'
                };

                rescuer.tasks.push(taskCopy); // Add to rescuer's tasks list
            }
        });

        // Save the updated civilian and rescuer documents
        await civilian.save();
        await rescuer.save();

        res.json({ success: true, message: 'Tasks served successfully and updated for both civilian and rescuer.' });

            'tasks._id': { $in: taskIds } // Find civilians who have the selected task IDs
        }, 'tasks location'); // Fetch their location and tasks only
        console.log(`\nCivilians found: ${civilians}\n`);
        

        if (!civilians || civilians.length === 0) {
            return res.status(404).json({ success: false, message: 'No civilians found for the selected tasks' });
        }
        

        // 3. Loop through civilians to get the tasks and their location
        civilians.forEach(civilian => {
            const civilianLat = civilian.location.coordinates[0]; // Latitude
            const civilianLng = civilian.location.coordinates[1]; // Longitude

            // Calculate the distance between the rescuer and the civilian
            const distance = haversineDistance(
                rescuerLocation.lat,
                rescuerLocation.lng,
                civilianLat,
                civilianLng
            );

            console.log(`Distance between rescuer and civilian: ${distance} meters`);

            // Check if the task is within 50 meters
            if (distance <= 50) {
                // Filter out the task IDs that match and are within distance
                civilian.tasks.forEach(task => {
                    if (taskIds.includes(task._id.toString())) {
                        tasksWithinDistance.push({
                            taskId: task._id,
                            civilianId: civilian._id
                        });
                    }
                });
            }
        });

        // 4. Process tasks that are within the distance
        if (tasksWithinDistance.length > 0) {
            console.log('Tasks within 50 meters:', tasksWithinDistance);

            for (const task of tasksWithinDistance) {
                const { taskId, civilianId } = task;

                // Find the civilian by ID and update the corresponding task
                const civilian = await User.findById(civilianId);
                console.log(`civilian: ${civilian}`);
                if (civilian) {
                    civilian.tasks.forEach(civilianTask => {
                        if (civilianTask._id.toString() === taskId.toString()) {
                            civilianTask.status = 'delivered';
                            civilianTask.dateServed = new Date(); // Set current date as the delivery date
                        }
                    });

                    // Save the updated civilian document
                    await civilian.save();

                    // Remove the task from the rescuer's task list
                    rescuer.tasks = rescuer.tasks.filter(rescuerTask => rescuerTask._id.toString() !== taskId.toString());
                }
            }

            // Save the updated rescuer document
            await rescuer.save();

            res.json({ success: true, message: 'Tasks successfully delivered and updated' });
        } else {
            console.log('No tasks within 50 meters');
            res.json({ success: true, message: 'No tasks within 50 meters' });
        }

    } catch (error) {
        console.error('Error delivering tasks:', error);
        res.status(500).json({ success: false, message: 'Internal server error' });
    }
});

router.get('/stats', async (req, res) => {
    const { startDate, endDate } = req.query;

    try {
        const matchDate = {
            $gte: new Date(startDate),
            $lte: new Date(endDate)
        };

        // Aggregate for total and completed requests
        const requestStats = await User.aggregate([
            { $unwind: "$tasks" },
            { $match: { "tasks.dateSubmitted": matchDate, "tasks.type": "request" } },
            {
                $group: {
                    _id: null,
                    totalRequests: { $sum: 1 },
                    completedRequests: {
                        $sum: {
                            $cond: [{ $eq: ["$tasks.status", "delivered"] }, 1, 0]
                        }
                    }
                }
            }
        ]);

        // Aggregate for total and completed donations
        const donationStats = await User.aggregate([
            { $unwind: "$tasks" },
            { $match: { "tasks.dateSubmitted": matchDate, "tasks.type": "donation" } },
            {
                $group: {
                    _id: null,
                    totalDonations: { $sum: 1 },
                    completedDonations: {
                        $sum: {
                            $cond: [{ $eq: ["$tasks.status", "delivered"] }, 1, 0]
                        }
                    }
                }
            }
        ]);

        // Send the results back as JSON
        res.json({
            totalRequests: requestStats[0]?.totalRequests || 0,
            completedRequests: requestStats[0]?.completedRequests || 0,
            totalDonations: donationStats[0]?.totalDonations || 0,
            completedDonations: donationStats[0]?.completedDonations || 0
        });

    } catch (err) {
        console.error('Error fetching stats:', err);
        res.status(500).send('Server error');
    }
});

<script>
    function openTab(tabId, element) {
        // Hide all tab contents
        const tabContents = document.querySelectorAll('.tab-content');
        tabContents.forEach(tabContent => tabContent.style.display = 'none');

        // Remove active class from all tab buttons
        const tabButtons = document.querySelectorAll('.tab-button');
        tabButtons.forEach(tabButton => tabButton.classList.remove('active'));

        // Show the clicked tab's content and mark the button as active
        document.getElementById(tabId).style.display = 'block';
        element.classList.add('active');

        // Check if it's the request tab and call generateTaskForm if it is
        if (tabId === 'requestTab') {
            generateTaskForm();  
        }
        else if (tabId === 'historyTab') {
            fetchTasks(); 
        }
        // Handle the announcements tab if needed
        else if (tabId === 'announcementTab') {
            fetchAnnouncements();
        }
    }

    // Open the Request tab by default and generate the form
    document.addEventListener('DOMContentLoaded', function () {
        document.getElementById('requestCheckbox').checked = true;  // Default to checked
        openTab('requestTab', document.getElementById('requestTabBtn'));  // Open Request tab
    });
    

    
</script>

<script>
    // Function to use Geolocation API and Reverse Geocoding
    function detectLocation() {
        if (navigator.geolocation) {
            navigator.geolocation.getCurrentPosition(successCallback, errorCallback);
        } else {
            alert("Geolocation is not supported by this browser.");
        }
    }

    function successCallback(position) {
        const latitude = position.coords.latitude;
        const longitude = position.coords.longitude;

        // Use ESRI Geocoder to reverse geocode coordinates to an address
        const geocodeService = L.esri.Geocoding.geocodeService();
        geocodeService.reverse().latlng([latitude, longitude]).run(function (error, result) {
            if (error) {
                alert("Unable to retrieve address.");
                return;
            }

            const fullAddress = result.address.Address || '';
            const city = result.address.City || '';
            const zip = result.address.Postal || '';
            const country = result.address.CountryCode || '';

            const addressParts = fullAddress.split(' ');
            const streetName = addressParts[0];
            const streetNumber = addressParts[1];

            const userConfirmed = confirm(`Is this your current location?\nStreet: ${streetName}\nNumber: ${streetNumber}\nCity: ${city}\nPostal Code: ${zip}\nCountry: ${country}`);
            if (userConfirmed) {
                document.getElementById('street').value = streetName;
                document.getElementById('number').value = streetNumber;
                document.getElementById('city').value = city;
                document.getElementById('zip').value = zip;
                document.getElementById('country').value = country;
            }
        });
    }

    function errorCallback(error) {
        alert("Error occurred while retrieving your location.");
    }

    window.onload = detectLocation;
</script>

    function applyFilter() {

        const isRequestChecked = document.querySelector('.sub-filter[value="request"]').checked;

            } else if (isDonationChecked && donationCount > requestCount) {
                baseIcon = createCombinedIcon(baseIcon, '/maptry/icons/donate.png');  // Add donation overlay
            }
    
            // Show or hide based on the task and main filter logic
            if (isPendingChecked && isTaskPending) {
                showMarker = true;
            } else if (isServedChecked && isTaskServed) {
                showMarker = true;
            }

            } else {

// Function to generate a random location within a 5km radius
function getRandomLocation(lat, lng, radiusInMeters = 5000) {
    const earthRadius = 6371000; // Earth's radius in meters

    const randomAngle = Math.random() * 2 * Math.PI; // Random angle in radians
    const randomDistance = Math.random() * radiusInMeters; // Random distance within the radius

    const deltaLat = randomDistance * Math.cos(randomAngle) / earthRadius;
    const deltaLng = randomDistance * Math.sin(randomAngle) / (earthRadius * Math.cos((Math.PI / 180) * lat));

    const newLat = lat + deltaLat * (180 / Math.PI);
    const newLng = lng + deltaLng * (180 / Math.PI);

    return [newLat, newLng];
}

// Function to check if the random location is on land using OpenStreetMap Nominatim API
async function isLand(lat, lng) {
    const response = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lng}&zoom=10`);
    const data = await response.json();

    // Check if the location has an address or road, indicating it is on land
    return data && data.address && data.address.road ? true : false;
}

// Function to initialize the rescuer at a random location
async function initRandomRescuerLocation(storageLat, storageLng, maxAttempts = 10) {
    let attempts = 0;
    let randomLocation;

    while (attempts < maxAttempts) {
        // Generate random location within 5km of storage
        randomLocation = getRandomLocation(storageLat, storageLng);

        // Check if the generated location is on land
        const isOnLand = await isLand(randomLocation[0], randomLocation[1]);

        if (isOnLand) {
            console.log(`Valid land location found: ${randomLocation}`);
            return randomLocation;
        }

        attempts++;
    }

    console.error("Failed to find valid land location after multiple attempts.");
    return null; // In case no valid land location is found
}

function renderItemsModal(items) {
       
    const itemsTable = document.getElementById('itemsTable');
    
    // Clear any existing rows in the table
    itemsTable.innerHTML = `
        <thead>
            <tr>
                <th>Item Name</th>
                <th>Category</th>
                <th>Details</th>
                <th>Stock</th>
                <th>Quantity</th>
                <th>Actions</th>
            </tr>
        </thead>
        <tbody></tbody>
    `;

    const tableBody = itemsTable.querySelector('tbody');

    // Loop through each product and create a row
    items.forEach(item => {
        const row = document.createElement('tr');
        row.dataset.itemId = item.id;
        row.dataset.itemName = item.name;   // Needed for search filter
        row.dataset.categoryName = item.category;   // Needed for search filter

        // Product Name
        const nameCell = document.createElement('td');
        nameCell.innerText = item.name;
        row.appendChild(nameCell);

        // Category
        const categoryCell = document.createElement('td');
        categoryCell.innerText = item.category;
        row.appendChild(categoryCell);

        // Details
        const detailsCell = document.createElement('td');
        detailsCell.innerText = item.details;
        row.appendChild(detailsCell);

        // Stock
        const stockCell = document.createElement('td');
        stockCell.innerText = item.stock;
        row.appendChild(stockCell);

        // Quantity Input Field
        const quantityCell = document.createElement('td');
        const quantityInput = document.createElement('input');
       
        quantityInput.type = 'number';
        quantityInput.value = 0;
        quantityInput.min = -1;
        quantityInput.max = item.stock || 0; // Fallback to 0 if stock is not defined
        quantityInput.style.width = '50px';
        
        let debounceTimer;
        quantityInput.addEventListener('input', function () {
            // Clear the previous timer whenever the user types
            clearTimeout(debounceTimer);

            // Set a new timer to delay the execution by 100 milliseconds (0.1 seconds)
            debounceTimer = setTimeout(function() {
                // Get the new value from the input
                let newQuantity = quantityInput.value;
                
                // Handle the empty string (if the input is deleted by the user)
                if (!isNaN(newQuantity) && parseInt(newQuantity) > 0) {
                    // If the input is a valid number and >= 0, update the modal
                    updateSelectedItemsModal(item, parseInt(newQuantity, 10));  // Convert the valid input to integer and update
                }
                else {
                    // Reset the value to empty if the input is invalid (e.g., non-numeric)
                    quantityInput.value = '';
                    updateSelectedItemsModal(item, 0);
                }
            }, 700);  // Delay execution by 100 milliseconds (0.1 seconds)
        });
        quantityCell.appendChild(quantityInput);
        row.appendChild(quantityCell);

        // Increment/Decrement Buttons
        const actionsCell = document.createElement('td');
        const incrementButton = document.createElement('button');
        incrementButton.innerText = '+';
        incrementButton.addEventListener('click', function () {
            let currentQuantity = parseInt(quantityInput.value);
            if (currentQuantity < item.stock) {
                quantityInput.value = currentQuantity + 1;
                updateSelectedItemsModal(item, 1);
            }
        });

        const decrementButton = document.createElement('button');
        decrementButton.innerText = '-';
        decrementButton.addEventListener('click', function () {
            let currentQuantity = parseInt(quantityInput.value);
            if (currentQuantity > 0) {
                quantityInput.value = currentQuantity - 1;
                updateSelectedItemsModal(item, -1);
            }
        });

        // Append buttons to the actions cell
        actionsCell.appendChild(incrementButton);
        actionsCell.appendChild(decrementButton);
        row.appendChild(actionsCell);

        // Add the row to the table body
        tableBody.appendChild(row);
    });

    // Call the filter function after rendering the items
    filterItemsModal();
}

// Function 5: Checkbox for editing the cargo items
function addCheckboxEventListeners() {
    document.querySelectorAll('.item-checkbox').forEach(checkbox => {
        checkbox.addEventListener('change', function() {
            const row = this.closest('tr');
            const quantityCell = row.querySelector('.quantity-cell');

            if (!quantityCell) {
                console.error('Quantity cell not found for row:', row);
                return;
            }

            // If the checkbox is checked, allow editing of quantity
            if (this.checked) {
                // Get the current quantity from the cell and store it
                let originalQuantity = parseInt(quantityCell.textContent, 10);
                if (isNaN(originalQuantity)) {
                    console.warn('Parsed quantity is NaN, defaulting to 0');
                    originalQuantity = 0;
                }
                // Store the original quantity in a data attribute for later use
                this.dataset.originalQuantity = originalQuantity;

                // Replace quantity text with an input field and buttons
                quantityCell.innerHTML = `
                    <input type="number" class="quantity-input" value="${originalQuantity}" min="0" style="width: 50px;">
                    <button class="increment-btn">+</button>
                    <button class="decrement-btn">-</button>
                `;

                // Add event listeners to the increment and decrement buttons
                const incrementBtn = quantityCell.querySelector('.increment-btn');
                const decrementBtn = quantityCell.querySelector('.decrement-btn');
                const quantityInput = quantityCell.querySelector('.quantity-input');

                incrementBtn.addEventListener('click', function() {
                    let currentQuantity = parseInt(quantityInput.value, 10);
                    quantityInput.value = currentQuantity + 1;
                });

                decrementBtn.addEventListener('click', function() {
                    let currentQuantity = parseInt(quantityInput.value, 10);
                    if (currentQuantity > 0) {
                        quantityInput.value = currentQuantity - 1;
                    }
                });

            } else {
                // If the checkbox is unchecked, restore the original quantity
                const originalQuantity = this.dataset.originalQuantity || 0;
                quantityCell.innerHTML = originalQuantity;
            }
        });
    });
}



// Function 6: Checkboxes @ Cargo editing - observing function
function observeCheckboxes() {
    document.addEventListener("DOMContentLoaded", function() {
        const targetNode = document.querySelector('#loadedItems tbody'); // Assuming checkboxes are within <tbody>
          
        if (!targetNode) {
            console.error("Target node (tbody) not found. Ensure it exists before observing.");
            return;
        }

        const config = { childList: true, subtree: true };

        const callback = function(mutationsList) {
            for (let mutation of mutationsList) {
                if (mutation.type === 'childList' && mutation.addedNodes.length > 0) {
                    addCheckboxEventListeners();
                }
            }
        };

        const observer = new MutationObserver(callback);
        observer.observe(targetNode, config);
    });
}

    }

function applyFilter() {
    const isPendingChecked = document.querySelector('.main-filter[value="pending"]').checked;
    const isServedChecked = document.querySelector('.main-filter[value="served"]').checked;
    const isRequestChecked = document.querySelector('.sub-filter[value="request"]').checked;
    const isDonationChecked = document.querySelector('.sub-filter[value="donation"]').checked;
    const isRouteChecked = document.getElementById('routeFilter').checked;

    const servedCivilianLocations = [];
    const rescuerWaypoints = [];

    // Loop through civilian markers to update icons and visibility
    for (let civilianId in civilianMarkers) {
        const { marker, tasks } = civilianMarkers[civilianId];
        let showMarker = false;
        let baseIcon = CivilianPendingIcon;

        // Check if the marker is valid and hasn't been removed already
        

        const hasProcessedTask = tasks.some(task => task.status === 'processed');
        const hasOnlyDonateTasks = tasks.every(task => task.type === 'donation');

        if (hasProcessedTask) {
            baseIcon = CivilianServedIcon;
            if (isServedChecked) showMarker = true;
            servedCivilianLocations.push(marker.getLatLng());
        } else if (hasOnlyDonateTasks) {
            baseIcon = CivilianDonationIcon;
            if (isPendingChecked) showMarker = true;
        } else {
            baseIcon = CivilianPendingIcon;
            if (isPendingChecked) showMarker = true;
        }

        const requestCount = tasks.filter(task => task.type === 'request').length;
        const donateCount = tasks.filter(task => task.type === 'donation').length;

        if (isRequestChecked && requestCount > donateCount) {
            baseIcon = createCombinedIcon(baseIcon, '/maptry/icons/req.png');
        } else if (isDonationChecked && donateCount >= requestCount) {
            baseIcon = createCombinedIcon(baseIcon, '/maptry/icons/donate.png');
        }

        // Update the marker icon
        marker.setIcon(baseIcon);

        // Only add to the map if the marker should be visible
        
    }

    // Loop through rescuer markers to update visibility
    for (let rescuerId in rescuerMarkers) {
        const { marker, pendingTasks } = rescuerMarkers[rescuerId];
        let showMarker = false;

        // Check if the marker is valid and hasn't been removed already
        

        if (isPendingChecked && pendingTasks.length === 0) {
            showMarker = true; // Show rescuers with no tasks if pending is checked
        } else if (isServedChecked && pendingTasks.length > 0) {
            showMarker = true; // Show rescuers with tasks if served is checked
            rescuerWaypoints.push(marker.getLatLng());
        }

        // if (showMarker && !map.hasLayer(marker)) {
        //     marker.addTo(map);
        // } else if (!showMarker && map.hasLayer(marker)) {
        //     marker.remove();
        // }
    }

    // Handle Route Drawing
    if (isRouteChecked && isServedChecked && servedCivilianLocations.length > 0 && rescuerWaypoints.length > 0) {
        drawRouteBetweenMarkers(rescuerWaypoints, servedCivilianLocations);
    } else if (routingControl) {
        map.removeControl(routingControl);
    }
}

function debounce(func, wait) {
    let timeout;
    return function(...args) {
        clearTimeout(timeout);
        timeout = setTimeout(() => func.apply(this, args), wait);
    };
}

            markerCivilian.on('moveend', debouncedUpdateAddress);
        
        } else {
            console.error("No location data found for civilian.");
        }
    } catch (error) {
        console.error('Error fetching civilian location:', error);
    }
};

// Debounced update function for marker movement
const debouncedUpdateAddress = debounce(async function() {
    if (markerCivilian) {
        const latLng = markerCivilian.getLatLng();
        await populateAddress();
    
        try {
            const response = await fetch('/api/civilian/updateLocation', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    lat: latLng.lat,
                    lng: latLng.lng
                })
            });
            // Ensure the response status is OK
            if (!response.ok) {
                throw new Error('Failed to update location');
            }

            const data = await response.json();
            console.log('Location update response:', data); // Log response

            // Update success/failure message based on data
            if (data.message) {
                console.log(data.message); // Log the success message
            } else {
                console.error('Failed to update location: No message returned');
            }
        } catch (error) {
            console.error('Error:', error.message || 'Failed to update location');
        }
    }
}, 300); // 300ms debounce delay

async function populateItemsAndCategories() {
    try {
        const response = await fetch('/api/items');  // Fetch from the new Data-based route
        const data = await response.json();
        
        const categories = Array.from(new Set(data.map(item => item.category)));
        const categorySelect = document.getElementById('category');
        const itemSelect = document.getElementById('itemName');
        
        // Populate categories for user selection
        categorySelect.innerHTML = '<option value="">All Categories</option>'; // Default to all categories
        categories.forEach(category => {
            if (category) {  // Ensure category is valid
                const option = document.createElement('option');
                option.value = category;  // Use category as value
                option.text = category;  // Display category name
                categorySelect.appendChild(option);
            }
        });

        // Populate items when a category is selected
        categorySelect.addEventListener('change', function() {
            const selectedCategory = categorySelect.value;
            itemSelect.innerHTML = ''; // Clear the item select options

            // Filter items based on the selected category
            const filteredItems = selectedCategory 
                ? data.filter(item => item.category === selectedCategory)
                : data;  // Show all items if no category selected

            filteredItems.forEach(item => {
                if (item && item.id && item.name && item.category) {  // Ensure item has valid properties
                    const option = document.createElement('option');
                    option.value = item.id;
                    option.text = item.name;
                    option.setAttribute('data-category', item.category);  // Attach the category id to the item
                    itemSelect.appendChild(option);
                }
            });
        });

        // Automatically populate the items with no category filter by default
        data.forEach(item => {
            if (item && item.id && item.name && item.category) {  // Ensure item has valid properties
                const option = document.createElement('option');
                option.value = item.id;
                option.text = item.name;
                option.setAttribute('data-category', item.category);  // Attach the category id to the item
                itemSelect.appendChild(option);
            }

// Function to submit task
async function submitTask() {
    const taskType = document.getElementById('taskType').value;
    const itemSelect = document.getElementById('itemName');
    
    // Ensure itemSelect exists and has a value
    if (!itemSelect || !itemSelect.value || !civilianId) {
        console.error("Item select element, its value, or civilianId is missing.");
        return;
    }

    const itemId = itemSelect.value;  // The selected item id
    
    const selectedItemOption = itemSelect.options[itemSelect.selectedIndex];
    const itemName = selectedItemOption.textContent;  // Get the item name
    // Check if the selected option has the 'data-category' attribute
    if (!selectedItemOption || !selectedItemOption.getAttribute('data-category')) {
        console.error("No valid item selected or item is missing its category.");
        return;
    }

    const itemCategory = selectedItemOption.getAttribute('data-category');
    const quantity = document.getElementById('quantity').value;

    if (!itemCategory) {
        alert('Item category is missing. Please try selecting the item again.');
        return;
    }

    const street = document.getElementById('street').value;
    const number = document.getElementById('number').value;
    const city = document.getElementById('city').value;
    const country = document.getElementById('country').value;
    const zip = document.getElementById('zip').value;

    const taskData = {
        type: taskType,
        status: 'pending',
        civilianId: civilianId,
        address: { street, number, city, zip, country },
        item: itemName,  // Pass the selected item id
        itemCategory: itemCategory,  // Use the category id associated with the selected item
        quantity,
        rescuerUsername: null, 
        dateServed: null 
    };

    console.log("Submitting task data:", taskData);

    try {
        const response = await fetch('/api/civilian/tasks', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify(taskData),
        });

        if (response.ok) {
            alert('Task submitted successfully!');
        } else {
            const errorData = await response.json();
            console.error('Error submitting task:', errorData);
            alert('Error submitting task: ' + errorData.message);
        }
    } catch (error) {
        console.error('Error submitting task:', error);
        alert('Error submitting task');
    }
}

            url: `/api/stats/chart-data`,
            method: 'GET',
            data: { startDate: startDate, endDate: endDate },
            success: function (data) {
                renderChart(data);
            },
            error: function (error) {
                console.error('Error fetching chart data:', error);
            }
        });
    }
    
    // Function to render chart
    function renderChart(data) {
        const ctx = document.getElementById('task-chart').getContext('2d');
        
        // Check if a chart instance exists and destroy it before creating a new one
        if (chartInstance) {
            chartInstance.destroy();
        }
        
        // Create new chart instance
        chartInstance = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: ['Total Requests', 'Total Donations', 'Completed Requests', 'Completed Donations'],
                datasets: [{
                    label: 'Amount',
                    data: [data.totalRequests, data.totalDonations, data.completedRequests, data.completedDonations],
                    backgroundColor: [
                        'rgba(255, 99, 132, 0.2)',
                        'rgba(54, 162, 235, 0.2)',
                        'rgba(255, 206, 86, 0.2)',
                        'rgba(75, 192, 192, 0.2)',
                    ],
                    borderColor: [
                        'rgba(255, 99, 132, 1)',
                        'rgba(54, 162, 235, 1)',
                        'rgba(255, 206, 86, 1)',
                        'rgba(75, 192, 192, 1)',
                    ],
                    borderWidth: 1
                }]
            },
            options: {
                scales: {
                    y: {
                        beginAtZero: true
                    }
                }
            }
        });
    }

    // Event listener for updating chart based on selected date range
    $('#updateChart').on('click', function () {
        const startDate = $('#startDate').val(); // Corrected ID
        const endDate = $('#endDate').val();     // Corrected ID
        
        if (startDate && endDate) {
            const startISO = new Date(startDate).toISOString();
            const endISO = new Date(endDate).toISOString();
            
            fetchChartData(startISO, endISO);
        } else {
            alert('Please select both start and end dates.');
        }
    });

$(document).ready(function () {
    function checkLowStock() {
        $.ajax({
            url: '/api/items/low-stock',  // The backend route for low stock items
            type: 'GET',
            success: function(response) {
                // If there are low stock items, display an alert
                if (response.lowStockItems && response.lowStockItems.length > 0) {
                    let alertMessage = 'Warning: The following items are low in stock:\n';
                    
                    // Loop through the low stock items and add them to the alert message
                    response.lowStockItems.forEach(itemData => {
                        itemData.items.forEach(item => {
                            alertMessage += `Item Name: ${item.name}, Stock: ${item.stock}\n`;
                        });
                    });
                    
                    alert(alertMessage);
                } else if (response.message) {
                    console.log(response.message);  // Log if no low stock items
                }
            },
            error: function(error) {
                console.error('Error fetching low stock items:', error);
            }
        });
    }

    // Check for low stock items when the page is loaded
    checkLowStock();
    
    

});

const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

// Base user schema
const baseOptions = {
    discriminatorKey: 'role', // Define the key for differentiating user types
    collection: 'users', // Ensure all user types go into the same collection
};

const baseUserSchema = new mongoose.Schema({
    username: { type: String, required: true, unique: true },
    email: { type: String, required: true, unique: true },
    password: { type: String, required: true },
    location: {
        type: { type: String, enum: ['Point'], default: 'Point' }, 
        coordinates: { type: [Number], default: [38.24600, 21.73500] } // Default coordinates
    }
}, baseOptions);

// Encrypt password before saving
baseUserSchema.pre('save', async function (next) {
    if (!this.isModified('password')) return next();
    this.password = await bcrypt.hash(this.password, 12);
    next();
});

// Password comparison method
baseUserSchema.methods.correctPassword = async function (candidatePassword, userPassword) {
    return await bcrypt.compare(candidatePassword, userPassword);
};

// Method for adding tasks (shared by all roles)
baseUserSchema.methods.addTask = async function (task) {
    this.tasks.push(task);
    await this.save();
};

// Index location if needed for geospatial queries
baseUserSchema.index({ location: '2dsphere' });

const taskSchema = new mongoose.Schema({
    type: { type: String, enum: ['request', 'donation'], required: true },
    status: { type: String, enum: ['pending', 'processed', 'delivered'], default: 'pending' },
    civilianId: { type: mongoose.Schema.Types.ObjectId, ref: 'User', required: true }, // Reference to the Civilian
    address: {
        street: String,
        number: String,
        city: String,
        zip: String,
        country: String,
    },
    item: { type: String, ref: 'Item', required: true },
    itemCategory: { type: String, ref: 'Category', required: true }, // Reference to Category
    quantity: { type: Number, required: true },
    dateSubmitted: { type: Date, default: Date.now },
    rescuerUsername: { type: mongoose.Schema.Types.ObjectId, ref: 'User'}, // Filled when processed
    dateServed: { type: Date }

})

// Civilian schema (inherits from baseUserSchema)
const civilianSchema = new mongoose.Schema({
    name: { type: String, required: true },
    cellphone: { type: String, required: true },
    address: {
        street: String,
        number: String,
        city: String,
        zip: String,
        country: String,
    },
    tasks: [taskSchema]
});

// Rescuer schema (inherits from baseUserSchema)
const rescuerSchema = new mongoose.Schema({
    cargo: [
        {
            item: { type:String, ref: 'Data', required: true },
            itemCategory: { type: String, ref: 'Category', required: true },
            quantity: { type: Number, required: true }
        }
    ],
    tasks: [taskSchema]
});

// Admin schema (inherits from baseUserSchema)
const adminSchema = new mongoose.Schema({
    announcements: [
        {
            type: [{
                date: { type: String },
                announcement: { type: String },
                items: { type: String }
            }]
        }
    ]
});

// Create the base model
const User = mongoose.model('User', baseUserSchema);

// Define role-specific models using discriminators
const Civilian = User.discriminator('civilian', civilianSchema);
const Rescuer = User.discriminator('rescuer', rescuerSchema);
const Admin = User.discriminator('admin', adminSchema);

module.exports = {
    User,
    Civilian,
    Rescuer,
    Admin
};

const mongoose = require('mongoose');
const Schema = mongoose.Schema;

// Define the details schema for items
const DetailSchema = new Schema({
  detail_name: { type: String, default: null },
  detail_value: { type: String, default: null }
});

// Define the categories schema
const CategorySchema = new Schema({
  id: { type: String, required: true },
  category_name: { type: String, default: null }
});

// Define the items schema with a reference to categories
const ItemSchema = new Schema({
  id: { type: String, required: true },
  name: { type: String, default: null },
  category: { type: Schema.Types.String, ref: 'Category', required: true, default: null}, // Reference to category id
  details: { type: [DetailSchema], default: null }, // Embedded array of details
  stock: { type: Number, default: null }
});

// Define the main schema containing both items and categories
const DataSchema = new Schema({
  items: [ItemSchema],
  categories: [CategorySchema]
});

// Create the models
const Item = mongoose.model('Item', ItemSchema);
const Category = mongoose.model('Category', CategorySchema);
const Data = mongoose.model('Data', DataSchema);

module.exports = { Item, Category, Data };

exports.updateCategory = async (req, res) => {
    const { id, category_name } = req.body;  // New ID and category name

    try {
        const categoriesDoc = await Category.findOne();  // Fetch the first document

        // Find the category by its original id from the params
        const categoryIndex = categoriesDoc.categories.findIndex(cat => cat.id === req.params.id);

        if (categoryIndex === -1) {
            return res.status(404).json({ error: 'Category not found' });
        }

        // Check if the new ID already exists in another category
        const existingCategory = categoriesDoc.categories.find(cat => cat.id === id && cat.id !== req.params.id);
        if (existingCategory) {
            return res.status(400).json({ error: 'The new ID already exists. It must be unique.' });
        }

        // Update the category at the found index
        categoriesDoc.categories[categoryIndex].id = id;
        categoriesDoc.categories[categoryIndex].category_name = category_name;

        await categoriesDoc.save();
        res.json({ message: 'Category updated successfully', category: categoriesDoc.categories[categoryIndex] });
    } catch (err) {
        res.status(400).json({ error: err.message });
    }
};

exports.deleteCategory = async (req, res) => {
    try {
        const categoriesDoc = await Category.findOne();  // Fetch the first document

        // Find the category by its id
        const categoryIndex = categoriesDoc.categories.findIndex(cat => cat.id === req.params.id);

        if (categoryIndex === -1) {
            return res.status(404).json({ error: 'Category not found' });
        }

        // Remove the category from the array
        categoriesDoc.categories.splice(categoryIndex, 1);

        await categoriesDoc.save();
        res.json({ message: 'Category deleted successfully' });

    }
};
